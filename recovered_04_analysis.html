Description of data variables:

Sequencing identifier associated with the sequencing samples. They look like sequencing run and barcode information Time (4h and 24h) Stimulation (Candita/ Aspergillus / Rhizopus / RPMI(control group)) Samples (8 samples that represent 8 biological replicates/ 8 different human donours) Gene expression data (The nymbers represent the number of sequencing reads assigned to each gene in each sample)

Tidy the dataset:

1) Extract sample metadata

metadata_raw <- as.data.frame(raw_data)
names(metadata_raw)[1] <- "parameter"
metadata_raw <- metadata_raw[1:3, ]  # extract first 3 row with all metadata

2) Transform the metadata from wide format to long format

metadata_long <- metadata_raw |>
  pivot_longer(
    cols = -parameter,
    names_to = "sample_id",
    values_to = "value"
  )
# reorder metadata_long
metadata_long <- metadata_long |>
  select(sample_id, parameter, value)

3) Back to wide, to have one row per sample

metadata <- metadata_long |>
  pivot_wider(
    names_from = parameter,
    values_from = value
  )

metadata <- metadata |>
  rename_with(~ "donor", "sample") |>
  rename_with(~ "stimulus", "stimu")

4) Renaming Stimulus column

metadata <- metadata |> 
  mutate(
    stimulus = recode(stimulus,
      "RPMI"                  = "Control_group",
      "Candida-albicans"      = "C_albicans",
      "Aspergillus-fumigatus" = "A_fumigatus",
      "Rhizopus-oryzae"       = "R_oryzae"),
    time = factor(time, levels = c("4h", "24h"))
  )

Collecting results to a dataframe:

res_df_4h <- as.data.frame(res_4h) |>
  rownames_to_column("gene_id")

Asigning log expression description of value up, down, not significant

res_df_4h <- res_df_4h |>
  mutate(
    significance = case_when(
      !is.na(padj) & padj < 0.05 & log2FoldChange >= 1 ~ "Up",
      !is.na(padj) & padj < 0.05 & log2FoldChange <= -1 ~ "Down",
      TRUE ~ "Not significant"
    )
  )
